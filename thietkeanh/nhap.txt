SELECT service_id, GROUP_CONCAT( CONCAT( CAST( licence_time_value AS CHAR ) , ' ', licence_time_unit ) )
FROM service_licence_time
LEFT OUTER JOIN licence_time ON licence_time.licence_time_id = service_licence_time.licence_time_id
WHERE service_id =2
GROUP BY service_id
